pokext Cheat Sheet v0.4.4

External memory patcher for Linux x86-64. Two APIs: raw — the injected target object, no import needed, and pokexhl — a Frida-shaped high-level wrapper. Mix them freely; they read the same memory.

Setup: 
# install once
git clone https://github.com/<you>/pokext.git
cd pokext && ./install.sh

# run any script
sudo pokext $(pidof -s <target>) script.py

1. Raw API no import

Every script gets a target object in its globals. No import, no wrapping — target.read(), target.write(), etc. work directly.

Symbols

# int32 by symbol name — implicit type
val = target.score
target.score = 1337

# find an address explicitly
addr = target.find("add_score")   # raises KeyError if missing

Memory access

CallReturns
target.read(addr, "i32")int (see type table)
target.write(addr, 2.5, "f32")—
target.read_ptr(addr)int (8-byte pointer)
target.read_bytes(addr, 16)bytes
target.write_bytes(addr, b"\x90")—
target.read_cstr(addr, 64)str

Scanning

# first match or None
hit = target.scan("48 8B 05 ?? ?? ?? ??")

# every match — list of ints
hits = target.scan_all("DE AD BE EF")

# restrict to one module
hit = target.scan("4? 8B ??", module="libc.so.6")
Wildcards: ?? matches any byte, 4? matches any low nibble.

Calls

# no args
target.call("GetFPS", ())

# int args
target.call("SetTargetFPS", (240,))

# return as float
target.call("GetFrameTime", (), ret="f32")

# struct-by-value args (SysV AMD64 registers)
target.call_struct("DrawSphere", (
    ("v3", 10.0, 5.0, 10.0),     # Vector3 → XMM0, XMM1
    ("f",  3.0),                 # float   → XMM2
    ("i",  0xFF0000FF),          # Color   → RDI
))
TagMeaningRegisters
("i", n)single integerRDI, RSI, ...
("f", n)single floatXMM0..7
("v2", x, y)two floats1 XMM
("v3", x, y, z)three floats2 XMM
("v4", x, y, z, w)four floats2 XMM

Hooks

# simple redirect — replacement runs instead of original
src, dst = target.hook("add_score", "add_score_boost")

# inline hook — returns a callable original via trampoline
tramp = target.hook_inline("add_score", "add_score_boost", prologue=16)
target.call(hex(tramp), (10,))   # run the ORIGINAL
target.unhook("add_score")

# raw byte patch (works on .text via ptrace)
target.patch(addr, b"\x90" * 8)

# list active
for h in target.hooks():
    print(h["name"], hex(h["addr"]), "->", hex(h["replacement"]))

Modules & meta

target.pid      # int
target.base     # int, ELF load base
target.exe      # str

for m in target.modules():
    print(m.name, hex(m.base), hex(m.size), m.path)

lib  = target.module("libc.so.6")
addr = lib.symbol("malloc")

# memory regions from /proc/pid/maps
for r in target.regions():
    print(hex(r["start"]), r["perms"], r["path"])
Multi-process: other = pokex.attach(6312) opens a second target. Same API on the returned object.

2. pokexhl high-level API

Frida-shaped wrapper. Pointers are objects, modules have symbols, functions are callable. Same underlying memory as the raw API. 

import pokexhl as px

p = px.Process()

Hooks

orig = p.hook("add_score", "add_score_boost")

p.hooks()

Watch

w.stop()       # later

Allocate in target

buf = p.alloc(4096)       # RWX mmap inside the target
buf.write_cstr("hello")

3. Both in one file combined

# example.py — raw + high-level API mixed
# run: sudo pokext $(pidof -s <target>) example.py
import time
import struct

import pokexhl as px


# ==================== raw API ====================
# `target` is injected into globals by pokext
# No import needed, no wrapper

print("--- raw API ---")

try:
    score_addr = target.find("score")
    print(f"score @ 0x{score_addr:x}")
    print(f"score = {target.read(score_addr, 'i32')}")

    target.write(score_addr, 9999, "i32")
    print(f"after write = {target.read(score_addr, 'i32')}")
except KeyError as e:
    print(f"symbol not found: {e}")

# raw scan
hits = target.scan_all("48 8B 05 ?? ?? ?? ??")
print(f"raw scan: {len(hits)} hits")


# ==================== high-level API ====================

print("\n--- high-level API ---")

p = px.Process()

print(f"attached to {p.exe} (pid {p.pid})")
print(f"base: {p.base}")

# same operations, object style
try:
    score = p.sym("score")
    print(f"score @ {score}")
    print(f"score = {score.read_i32()}")

    score.write_i32(7777)
    print(f"after write = {score.read_i32()}")
except KeyError as e:
    print(f"symbol not found: {e}")

# call into target
try:
    fps = p.fn("GetFPS")()
    print(f"GetFPS = {fps}")
except KeyError:
    pass


# ==================== both together ====================

print("\n--- mixed ---")

# build a pointer chain with pokexhl
try:
    player_ptr = p.sym("player")
    hp_addr    = player_ptr.follow(0x10, 0x8)

    # read the same memory both ways
    hp_hl  = hp_addr.read_i32()
    hp_raw = target.read(hp_addr.addr, "i32")
    print(f"health: hl={hp_hl}  raw={hp_raw}  match={hp_hl == hp_raw}")
except KeyError:
    pass

# raw scan, then wrap results as NativePointers
raw_hits = target.scan_all("DE AD BE EF")
wrapped  = [p.ptr(h) for h in raw_hits]
print(f"found {len(wrapped)} hits, first = {wrapped[0] if wrapped else 'none'}")


# ==================== practical cheat ====================

print("\n--- cheat: pin score at 999999 ---")

try:
    score = p.sym("score")
    print("press Ctrl-C to stop\n")

    while True:
        score.write_i32(999999)
        current = target.read(score.addr, "i32")
        print(f"\rscore = {current}", end="", flush=True)
        time.sleep(0.1)
except KeyboardInterrupt:
    print("\nstopped")
except KeyError:
    print("no 'score' symbol — try the scan version")
When to use which: raw for quick pokes and REPL exploration; pokexhl for anything longer than five lines. They read the same memory, so mix freely.

Common Patterns

Scan and pin

import struct, time
import pokexhl as px

p = px.Process()

v = int(input("current value: "))
hits = p.scan(struct.pack("<i", v).hex(" ")).all()
print(f"{len(hits)} candidates")

input("change it, then Enter")
new = int(input("new value: "))

refined = [h for h in hits if h.read_i32() == new]
addr = refined[0]
print(f"pinning at 9999")

try:
    while True:
        addr.write_i32(9999)
        time.sleep(0.1)
except KeyboardInterrupt:
    pass

Read multiple struct fields

import pokexhl as px
p = px.Process()

player = p.sym("player_entity")

hp    = player.add(0x40).read_i32()
armor = player.add(0x44).read_i32()
xp    = target.read(player.addr + 0x48, "i32")   # raw for variety

print(f"hp={hp} armor={armor} xp={xp}")

Watch and print

import time
import pokexhl as px

p = px.Process()
addr = p.sym("health")

w = p.watch(addr, "i32", interval=0.02)
w.on_change(lambda old, new: print(f"health: {old} -> {new}"))
w.start()

try:
    time.sleep(60)
except KeyboardInterrupt:
    w.stop()

Hook and log

import time
import pokexhl as px

p = px.Process()

try:
    orig = p.hook("add_score", "add_score_boost")
    print(f"hooked. original callable at {orig.addr:#x}")
    orig.call((10,))
    time.sleep(30)
finally:
    p.unhook("add_score")

Print on screen (via engine)

import time
import pokexhl as px

p = px.Process()
scratch = p.alloc(256)
fn = p.fn("SCR_CenterPrint")

while True:
    scratch.write_cstr("hello from pokext\x00")
    fn(scratch.addr)
    time.sleep(2)

Debugging

SymptomFix
KeyError: 'sym'Binary stripped → use scan
ModuleNotFoundError: pokexhlNot installed → ./install.sh
AttributeError: 'target' in imported moduleRebuild with the builtins fix
OSError: Bad addressptrace refused → run with sudo
Call hangs foreverTarget is blocking — don't call I/O or sleeps
Game crashes on hookBump prologue=32 in hook_inline
Values don't stickPage is read-only → use patch, not write
Legal. Use pokext on processes you own or have permission to inspect. Using it against software you don't control, or in online games with anti-cheat, will get you banned and may violate the game's terms of service.
